// setTimeout 模拟每一步耗时 const xzx_step = (msg, ms) => new Promise(res => setTimeout(() => res(msg), ms)); async function makeDumplings() { // ① 线上(6s)/线下(3s)同时买 —— 竞速，谁快用谁 await Promise.race([buyOnline(), buyOffline()]); // ② 取原料（2s） await takeIngredients(); // ③ 和面(5s) 与 和馅(4s) 同时进行 —— 全部完成才继续 await Promise.all([makeDough(), makeFilling()]); // ④⑤⑥ 包(3s) → 煮(4s) → 吃(2s) 依次串行 await wrapDumplings(); await boilDumplings(); await eatDumplings(); }
